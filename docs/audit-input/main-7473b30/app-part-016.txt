healthAction=async function(fn,pendingText='Kapcsolódás…'){
  if(!state.health)state.health={};
  if(state.health.busy){
    state.health.message='Egy másik Health Connect frissítés már fut. Várj néhány másodpercet, majd próbáld újra.';
    if(state.tab==='health'&&!state.session)render();
    return;
  }
  const tab=state.tab,page=typeof rf225HealthPage==='string'?rf225HealthPage:'hub';
  const stillHere=()=>tab==='health'&&state.tab===tab&&(!state.session)&&(typeof rf225HealthPage!=='string'||rf225HealthPage===page);
  const run=++rf228HealthRun;
  state.health.busy=true;
  state.health.busySince=Date.now();
  state.health.message=pendingText;
  if(stillHere())render();
  const watchdog=typeof setTimeout==='function'?setTimeout(()=>{
    if(run!==rf228HealthRun||!state.health.busy)return;
    rf228HealthRun++;
    state.health.busy=false;
    state.health.busySince=0;
    state.health.message='A Health Connect nem válaszolt időben. Próbáld újra, vagy ellenőrizd az Alvás és HRV engedélyeket.';
    if(stillHere())render();
  },20000):null;
  try{
    const msg=await fn(healthPlugin());
    if(run!==rf228HealthRun)return;
    state.health.message=msg||'Frissítés kész.';
  }catch(e){
    if(run!==rf228HealthRun)return;
    state.health.message=e?.message||'A Health Connect művelet nem sikerült.';
  }finally{
    if(watchdog!=null&&typeof clearTimeout==='function')clearTimeout(watchdog);
    if(run===rf228HealthRun){
      state.health.busy=false;
      state.health.busySince=0;
      if(stillHere())render();
    }
  }
};
